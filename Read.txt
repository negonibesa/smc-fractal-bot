risk_manager.py (добавить в класс)
python
def calculate_dynamic_risk(self, current_drawdown: float, pf_20: float, daily_pnl: float) -> float:
    """
    Рассчитывает динамический риск на основе состояния счета.
    """
    # 1. Множитель просадки (Drawdown)
    if current_drawdown < 0.03:         # < 3%
        dd_mult = 1.2
    elif current_drawdown < 0.06:       # < 6%
        dd_mult = 1.0
    else:                               # >= 6%
        dd_mult = 0.7
    
    # 2. Множитель PF (за последние 20 сделок)
    if pf_20 > 2.5:
        pf_mult = 1.2
    elif pf_20 > 1.5:
        pf_mult = 1.0
    else:
        pf_mult = 0.7
    
    # 3. Дневной PnL
    if daily_pnl < -0.02:               # дневной убыток > 2%
        daily_mult = 0.5
    else:
        daily_mult = 1.0
    
    # Итоговый риск
    risk = 1.5 * dd_mult * pf_mult * daily_mult
    
    # Ограничиваем
    risk = max(0.5, min(risk, 2.5))
    
    return round(risk, 2)
